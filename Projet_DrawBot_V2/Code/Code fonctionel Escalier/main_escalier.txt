#include <Arduino.h>
#include "config.h" // Ce fichier doit maintenant inclure INNER_WHEEL_SPEED_RATIO_ARC_TURN
#include <BluetoothSerial.h>

#if !defined(CONFIG_BT_ENABLED) || !defined(CONFIG_BLUEDROID_ENABLED)
#error Bluetooth is not enabled! Please run `make menuconfig` to and enable it
#endif

BluetoothSerial SerialBT;

// --- Initialisation des Broches et PWM ---
void setupMotors() {
  ledcSetup(PWM_CHANNEL_L_IN1, PWM_FREQ, PWM_RESOLUTION);
  ledcSetup(PWM_CHANNEL_L_IN2, PWM_FREQ, PWM_RESOLUTION);
  ledcSetup(PWM_CHANNEL_R_IN1, PWM_FREQ, PWM_RESOLUTION);
  ledcSetup(PWM_CHANNEL_R_IN2, PWM_FREQ, PWM_RESOLUTION);
  ledcAttachPin(IN_1_G_PIN, PWM_CHANNEL_L_IN1);
  ledcAttachPin(IN_2_G_PIN, PWM_CHANNEL_L_IN2);
  ledcAttachPin(IN_1_D_PIN, PWM_CHANNEL_R_IN1);
  ledcAttachPin(IN_2_D_PIN, PWM_CHANNEL_R_IN2);
  pinMode(EN_G_PIN, OUTPUT); pinMode(EN_D_PIN, OUTPUT);
  digitalWrite(EN_G_PIN, HIGH); digitalWrite(EN_D_PIN, HIGH);
  Serial.println("Moteurs et PWM initialisés.");
}

// --- Commande Individuelle d'un Moteur ---
void setMotorPower(int motor_id, int direction, int pwm_value) {
  pwm_value = constrain(pwm_value, 0, PWM_MAX);
  float correction_factor = 1.0;

  if (motor_id == LEFT_MOTOR) {
    correction_factor = LEFT_MOTOR_CORRECTION_FACTOR;
  } else if (motor_id == RIGHT_MOTOR) {
    correction_factor = RIGHT_MOTOR_CORRECTION_FACTOR;
  }
  int corrected_pwm = (int)(pwm_value * correction_factor);
  corrected_pwm = constrain(corrected_pwm, 0, PWM_MAX);

  Serial.print("setMotorPower -> ID: "); Serial.print(motor_id == LEFT_MOTOR ? "GAUCHE" : "DROIT");
  Serial.print(" | Dir: "); Serial.print(direction == FORWARD ? "AVANT" : (direction == BACKWARD ? "ARRIERE" : "STOP"));
  Serial.print(" | PWM (corrigé): "); Serial.println(corrected_pwm);

  if (motor_id == LEFT_MOTOR) {
    digitalWrite(EN_G_PIN, HIGH); 
    if (direction == FORWARD) {
      ledcWrite(PWM_CHANNEL_L_IN1, corrected_pwm); ledcWrite(PWM_CHANNEL_L_IN2, 0);
    } else if (direction == BACKWARD) {
      ledcWrite(PWM_CHANNEL_L_IN1, 0); ledcWrite(PWM_CHANNEL_L_IN2, corrected_pwm);
    } else { // STOP
      ledcWrite(PWM_CHANNEL_L_IN1, 0); ledcWrite(PWM_CHANNEL_L_IN2, 0);
    }
  } else if (motor_id == RIGHT_MOTOR) {
    digitalWrite(EN_D_PIN, HIGH); 
    if (direction == FORWARD) { // Logique inversée pour le moteur droit
      ledcWrite(PWM_CHANNEL_R_IN1, 0); ledcWrite(PWM_CHANNEL_R_IN2, corrected_pwm); 
    } else if (direction == BACKWARD) {
      ledcWrite(PWM_CHANNEL_R_IN1, corrected_pwm); ledcWrite(PWM_CHANNEL_R_IN2, 0); 
    } else { // STOP
      ledcWrite(PWM_CHANNEL_R_IN1, 0); ledcWrite(PWM_CHANNEL_R_IN2, 0);
    }
  }
}

// --- Arrêt d'Urgence / Arrêt Complet ---
void stopMotors() {
  Serial.println("Arrêt des moteurs.");
  setMotorPower(LEFT_MOTOR, STOP, 0);
  setMotorPower(RIGHT_MOTOR, STOP, 0);
}

// --- Fonctions de Mouvement Basées sur la Durée ---
void moveRobotStraight(int direction, int base_pwm_power, unsigned long duration_ms) {
  Serial.print("moveRobotStraight -> Dir: "); Serial.print(direction == FORWARD ? "AVANT" : "ARRIERE");
  Serial.print(" | PWM Base: "); Serial.print(base_pwm_power);
  Serial.print(" | Durée: "); Serial.println(duration_ms);

  setMotorPower(LEFT_MOTOR, direction, base_pwm_power);
  setMotorPower(RIGHT_MOTOR, direction, base_pwm_power);
  delay(duration_ms);
  stopMotors();
}

/**
 * @brief Fait tourner le robot en décrivant un ARC DE CERCLE pendant une durée définie.
 * Les deux roues tournent vers l'AVANT, mais la roue intérieure à l'arc tourne moins vite.
 * @param turn_direction TURN_LEFT ou TURN_RIGHT
 * @param base_pwm_power Puissance PWM de base (0-255) pour la roue EXTERIEURE
 * @param duration_ms Durée du mouvement en millisecondes
 */
void turnRobot(int turn_direction, int base_pwm_power, unsigned long duration_ms) {
  Serial.print("turnRobot (ARC) -> Dir: "); Serial.print(turn_direction == TURN_LEFT ? "GAUCHE" : "DROITE");
  Serial.print(" | PWM Base Ext: "); Serial.print(base_pwm_power);
  Serial.print(" | Ratio Int: "); Serial.print(INNER_WHEEL_SPEED_RATIO_ARC_TURN);
  Serial.print(" | Durée: "); Serial.println(duration_ms);

  int pwm_outer = base_pwm_power;
  int pwm_inner = constrain((int)(base_pwm_power * INNER_WHEEL_SPEED_RATIO_ARC_TURN), 0, PWM_MAX);

  if (turn_direction == TURN_LEFT) { // Pour tourner à GAUCHE (CCW), la roue DROITE est extérieure
    setMotorPower(RIGHT_MOTOR, FORWARD, pwm_outer);
    setMotorPower(LEFT_MOTOR, FORWARD, pwm_inner);
  } else if (turn_direction == TURN_RIGHT) { // Pour tourner à DROITE (CW), la roue GAUCHE est extérieure
    setMotorPower(LEFT_MOTOR, FORWARD, pwm_outer);
    setMotorPower(RIGHT_MOTOR, FORWARD, pwm_inner);
  }
  delay(duration_ms);
  stopMotors();
}

// --- Séquence Escalier ---
void sequenceEscalier() {
  Serial.println("--- DEBUT SEQUENCE ESCALIER (Contrôle par Durée, Virages en Arc) ---");
  SerialBT.println("Début Escalier (Durée, Arc)...");

  Serial.println("Escalier Etape 1: Avancer (simule 20cm)");
  moveRobotStraight(FORWARD, BASE_PWM_STRAIGHT, DURATION_FWD_20CM);
  delay(1000); 

  Serial.println("Escalier Etape 2: Tourner Gauche en ARC (simule 90deg)");
  turnRobot(TURN_LEFT, BASE_PWM_TURN, DURATION_TURN_90DEG); // Utilise la nouvelle fonction turnRobot
  delay(1000); 

  Serial.println("Escalier Etape 3: Avancer (simule 10cm)");
  moveRobotStraight(FORWARD, BASE_PWM_STRAIGHT, DURATION_FWD_10CM);
  delay(1000); 

  Serial.println("Escalier Etape 4: Tourner Droite en ARC (simule 90deg)");
  turnRobot(TURN_RIGHT, BASE_PWM_TURN, DURATION_TURN_90DEG); // Utilise la nouvelle fonction turnRobot
  delay(1000);  
  
  Serial.println("Escalier Etape 5: Avancer (simule 40cm)");
  moveRobotStraight(FORWARD, BASE_PWM_STRAIGHT, DURATION_FWD_40CM);             

  Serial.println("--- FIN SEQUENCE ESCALIER ---");
  SerialBT.println("Escalier (Durée, Arc) terminé.");
}

// --- Gestion des Commandes Bluetooth ---
void handleBluetoothCommand(String cmd) {
  cmd.trim();
  Serial.println("BT CMD RX: [" + cmd + "]");

  if (cmd.equalsIgnoreCase("S1")) {
    sequenceEscalier();
  } 
  else if (cmd.startsWith("FWD,")) {
    int first_comma = cmd.indexOf(',');
    int second_comma = cmd.indexOf(',', first_comma + 1);
    if (first_comma != -1 && second_comma != -1) {
      int pwm = cmd.substring(first_comma + 1, second_comma).toInt();
      unsigned long duration = cmd.substring(second_comma + 1).toInt();
      moveRobotStraight(FORWARD, pwm, duration);
    } else { SerialBT.println("Format: FWD,pwm,duree_ms"); }
  }
  else if (cmd.startsWith("BCK,")) {
    int first_comma = cmd.indexOf(',');
    int second_comma = cmd.indexOf(',', first_comma + 1);
    if (first_comma != -1 && second_comma != -1) {
      int pwm = cmd.substring(first_comma + 1, second_comma).toInt();
      unsigned long duration = cmd.substring(second_comma + 1).toInt();
      moveRobotStraight(BACKWARD, pwm, duration);
    } else { SerialBT.println("Format: BCK,pwm,duree_ms"); }
  }
  else if (cmd.startsWith("TL,")) { // Turn Left (Arc)
    int first_comma = cmd.indexOf(',');
    int second_comma = cmd.indexOf(',', first_comma + 1);
    if (first_comma != -1 && second_comma != -1) {
      int pwm = cmd.substring(first_comma + 1, second_comma).toInt();
      unsigned long duration = cmd.substring(second_comma + 1).toInt();
      turnRobot(TURN_LEFT, pwm, duration);
    } else { SerialBT.println("Format: TL,pwm,duree_ms"); }
  }
  else if (cmd.startsWith("TR,")) { // Turn Right (Arc)
    int first_comma = cmd.indexOf(',');
    int second_comma = cmd.indexOf(',', first_comma + 1);
    if (first_comma != -1 && second_comma != -1) {
      int pwm = cmd.substring(first_comma + 1, second_comma).toInt();
      unsigned long duration = cmd.substring(second_comma + 1).toInt();
      turnRobot(TURN_RIGHT, pwm, duration);
    } else { SerialBT.println("Format: TR,pwm,duree_ms"); }
  }
  else if (cmd.equalsIgnoreCase("STOP")) {
      stopMotors();
  }
  else {
    SerialBT.println("Commande inconnue. Options: S1, FWD,pwm,ms | BCK,pwm,ms | TL,pwm,ms | TR,pwm,ms | STOP");
  }
  SerialBT.println("Ready.");
}

// --- Fonctions Setup et Loop ---
void setup() {
  Serial.begin(115200);
  unsigned long setupStartTime = millis();
  while (!Serial && (millis() - setupStartTime < 2000)); 
  Serial.println("\nDrawbot Initialisation (Contrôle par Durée - Virages en Arc)...");

  setupMotors(); 
  stopMotors();  

  SerialBT.begin(BLUETOOTH_DEVICE_NAME);
  Serial.println("Bluetooth initialisé. Nom: " + String(BLUETOOTH_DEVICE_NAME));
  SerialBT.println("Drawbot (Durée, Arc) Prêt.");
  SerialBT.println("Cmds: S1 | FWD,pwm,ms | BCK,pwm,ms | TL,pwm,ms | TR,pwm,ms | STOP");
  Serial.println("Calibrez INNER_WHEEL_SPEED_RATIO_ARC_TURN et DURATION_TURN_90DEG dans config.h !");

  Serial.println("Setup terminé.");
}

void loop() {
  if (SerialBT.available()) {
    String command = SerialBT.readStringUntil('\n');
    handleBluetoothCommand(command);
  }
}
